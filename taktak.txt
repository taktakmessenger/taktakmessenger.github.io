# TakTak — prototipo (Node + React)

Resumen
TakTak es una app de vídeo corto (prototipo). Este repositorio contiene:
- Backend: Express (presigned uploads, feed, interacciones)
- Frontend: React (PWA/web) con player optimizado para feed vertical
- Infra local con Docker Compose (Postgres, Redis, MinIO, ffmpeg)

Quickstart (local)
1. Levantar infra:
   docker-compose up -d
2. Backend:
   cd taktak-api
   cp .env.example .env && ajustar variables
   npm install
   node index.js
3. Frontend:
   npx create-react-app taktak-web
   sustituir src/ con los archivos provistos
   cd taktak-web && npm install && npm start

Arquitectura recomendada
- Uploads: presigned S3 (MinIO en local) -> transcoding batch (ffmpeg) -> HLS/CDN
- Feed: eventos en Kafka/Redis Streams -> pipeline de features -> motor de recomendación
- Moderación: modelo ML + revisión humana
- ML/Generación: integrar MoneyPrinterTurbo / Pixelle-Video para features de generación
- Traducción/doblaje: KrillinAI

Licencias
Revisar licencias de los proyectos OSS antes de integrar (AGPL/GPL implican publicar cambios si incorporas/modify server code). Preferir adaptadores desacoplados.

Qué sigue
- Puedo generar Dockerfiles para backend/frontend y scripts de migración Postgres.
- Puedo sustituir los adaptadores por integraciones concretas (ej.: usar MoneyPrinterTurbo para generación).

const express = require('express');
const AWS = require('aws-sdk');
const bodyParser = require('body-parser');

const app = express();
app.use(bodyParser.json());

// Config S3 (MinIO compatible)
const s3 = new AWS.S3({
  endpoint: process.env.S3_ENDPOINT,
  accessKeyId: process.env.S3_ACCESS_KEY,
  secretAccessKey: process.env.S3_SECRET_KEY,
  s3ForcePathStyle: true,
  signatureVersion: 'v4',
  region: process.env.S3_REGION || 'us-east-1',
});

// In-memory store (prototipo). Sustituir por Postgres/Knex en producción.
const feed = []; // { id, user, title, videoKey, thumbnailKey, createdAt, likes }
let nextId = 1;

// Middleware JWT simple (stub)
function authMiddleware(req, res, next) {
  const auth = req.headers.authorization;
  if (!auth) return res.status(401).json({ error: 'missing auth' });
  const token = auth.split(' ')[1];
  try {
    req.user = jwt.verify(token, process.env.JWT_SECRET || 'change-me');
    next();
  } catch (e) {
    res.status(401).json({ error: 'invalid token' });
  }
}

app.get('/health', (req, res) => res.json({ ok: true }));

// Presigned URL for upload (PUT)
app.post('/presign', authMiddleware, async (req, res) => {
  const { filename, contentType } = req.body;
  if (!filename) return res.status(400).json({ error: 'filename required' });

  const key = `uploads/${Date.now()}_${filename}`;
  const params = {
    Bucket: process.env.S3_BUCKET,
    Key: key,
    Expires: 60 * 5, // 5 minutes
    ContentType: contentType || 'application/octet-stream',
  };
  try {
    const url = await s3.getSignedUrlPromise('putObject', params);
    return res.json({ url, key });
  } catch (err) {
    console.error('presign error', err);
    return res.status(500).json({ error: 'presign failed' });
  }
});

// Called after upload completes (frontend notifies)
app.post('/upload-complete', authMiddleware, (req, res) => {
  const { key, title } = req.body;
  if (!key) return res.status(400).json({ error: 'key required' });
  const item = {
    id: nextId++,
    user: req.user?.sub || 'anon',
    title: title || '',
    videoKey: key,
    thumbnailKey: '',
    createdAt: new Date().toISOString(),
    likes: 0,
  };
  feed.unshift(item);
  // En producción: push job a Redis/Bull para transcodificación y generación de thumbnails
  res.json({ ok: true, item });
});

// Simple feed endpoint
app.get('/feed', (req, res) => {
  const page = Number(req.query.page || 1);
  const per = Number(req.query.per || 10);
  const start = (page - 1) * per;
  const slice = feed.slice(start, start + per).map(i => ({
    id: i.id,
    title: i.title,
    videoUrl: `${process.env.S3_ENDPOINT}/${process.env.S3_BUCKET}/${i.videoKey}`,
    createdAt: i.createdAt,
    likes: i.likes,
  }));
  res.json({ data: slice, page });
});

// Interactions (like)
app.post('/interactions/like', authMiddleware, (req, res) => {
  const { id } = req.body;
  const item = feed.find(f => f.id === id);
  if (!item) return res.status(404).json({ error: 'not found' });
  item.likes = (item.likes || 0) + 1;
  res.json({ ok: true, likes: item.likes });
});

const port = process.env.PORT || 4000;
app.listen(port, () => console.log(`TakTak API listening on ${port}`));

import React, { useRef, useEffect } from 'react';

export default function VideoPlayer({ src }) {
  const ref = useRef(null);

  useEffect(() => {
    const v = ref.current;
    if (!v) return;
    const onVis = (entries) => {
      entries.forEach(e => {
        if (e.isIntersecting) { v.play().catch(()=>{}); }
        else { v.pause(); }
      });
    };
    const obs = new IntersectionObserver(onVis, { threshold: 0.5 });
    obs.observe(v);
    return () => obs.disconnect();
  }, []);

  return (
    <div style={{ width: '100%', position: 'relative', paddingTop: '177%' }}>
      <video
        ref={ref}
        src={src}
        style={{ position: 'absolute', top: 0, left: 0, width: '100%', height: '100%' }}
        muted
        playsInline
        controls
        preload="metadata"
      />
    </div>
  );
}


import React, { useRef } from 'react';

/**
 * Upload component:
 * - pide presigned URL al backend
 * - sube con PUT
 * - notifica al backend con key
 */
export default function Upload() {
  const inputRef = useRef();

  async function onFile(e) {
    const f = e.target.files[0];
    if (!f) return;
    // pedir presign
    const token = 'demo'; // Reemplazar por JWT auth real
    try {
      const pres = await axios.post('/presign', { filename: f.name, contentType: f.type }, { headers: { Authorization: `Bearer ${token}` } });
      const { url, key } = pres.data;
      // PUT directo
      await fetch(url, { method: 'PUT', headers: { 'Content-Type': f.type }, body: f });
      // notificar backend
      await axios.post('/upload-complete', { key, title: f.name }, { headers: { Authorization: `Bearer ${token}` } });
      alert('Upload completo — puede tardar procesamiento (transcode).');
    } catch (err) {
      console.error(err);
      alert('Error al subir');
    }
  }

  return (
    <div>
      <input ref={inputRef} type="file" accept="video/*" style={{ display: 'none' }} onChange={onFile} />
      <button onClick={() => inputRef.current.click()}>Subir vídeo</button>
    </div>
  );
}


# Guía de integración — Adaptadores OSS para TakTak

Resumen: aquí se documentan las opciones y recomendaciones para integrar proyectos OSS encontrados.

1) MoneyPrinterTurbo (https://github.com/harry0703/MoneyPrinterTurbo) — Generación de vídeos con AI
- Uso: pipeline de generación de contenido (auto-creación de shorts).
- Integración: exponer un worker que llame a MoneyPrinterTurbo (docker) para generar un video, subir resultado a S3 y crear entrada en feed.
- Licencia: MIT (permisiva).

2) douyin (https://github.com/zyronon/douyin) — Clon UI (Vue)
- Uso: referencia de UX para animaciones y preloading.
- Integración: no incluir código directamente si es GPL a menos que publiques cambios; mejor reimplementarlo (inspiración) o adaptar como microfrontend desacoplado.
- Licencia: GPL-3.0.

3) DKVideoPlayer (https://github.com/Doikki/DKVideoPlayer) — Player Android
- Uso: ideas de prefetch, buffering y filtros para versión Android.
- Integración: implementar estrategias similares en web y mobile native.

4) KrillinAI (https://github.com/krillinai/KrillinAI) — Traducción y doblaje de vídeo
- Uso: mejorar accesibilidad y alcance multilingüe.
- Integración: exponer endpoint /translate-video que llame a KrillinAI pipeline y luego actualice metadata.
- Licencia: GPL-3.0 — verificar obligaciones.

5) TikTokDownloader / tiktok-scraper / VidBee
- Uso: solo como herramientas de ingest/metadata para análisis, NO para replicar contenido sin permiso.
- Atención legal: respeto a TOS y derechos de autor.

6) Pixelle-Video / ShortGPT
- Uso: generación y edición automática para campañas de marketing o asistentes de creación de contenido.
- Integración: job que recibe un prompt y produce vídeo, luego presigned upload y publicación.

Notas de arquitectura y licencias
- Preferir adaptadores (servicios desacoplados) que consuman APIs de las soluciones OSS en contenedores propios.
- Evitar mezclar código AGPL/GPL en el backend principal; si lo incorporas, estarás obligado a publicar los cambios.
- Para cada integración, crea un contenedor independiente y define API estable (ingest/generate/status).

Checklist de integración práctica
- [ ] Crear contenedores Docker para cada OSS que quieras ejecutar.
- [ ] Añadir jobs en Redis/Bull para transcodificación y generación.
- [ ] Añadir monitorización y logs (Prometheus/ELK).
- [ ] Añadir tests y CI.

// Configurar base URL para axios (ej: REACT_APP_API_URL=http://localhost:4000)
axios.defaults.baseURL = process.env.REACT_APP_API_URL || '';

/* ----- Auth context ----- */
export const AuthContext = createContext({
  token: null,
  user: null,
  setToken: () => {},
  logout: () => {},
});

function useAuth() {
  const [token, setToken] = useState(() => localStorage.getItem('taktak_token'));
  const [user, setUser] = useState(() => {
    const raw = localStorage.getItem('taktak_user');
    try { return raw ? JSON.parse(raw) : null; } catch { return null; }
  });

  useEffect(() => {
    if (token) {
      localStorage.setItem('taktak_token', token);
      axios.defaults.headers.common.Authorization = `Bearer ${token}`;
    } else {
      localStorage.removeItem('taktak_token');
      delete axios.defaults.headers.common.Authorization;
    }
  }, [token]);

  const setAuth = (newToken, userObj = null) => {
    setToken(newToken);
    setUser(userObj);
    if (userObj) localStorage.setItem('taktak_user', JSON.stringify(userObj));
    else localStorage.removeItem('taktak_user');
  };

  const logout = () => {
    setToken(null);
    setUser(null);
  };

  return { token, user, setAuth, logout };
}

/* ----- Service Worker registration (PWA) ----- */
function registerServiceWorker() {
  if ('serviceWorker' in navigator) {
    navigator.serviceWorker
      .register('/service-worker.js')
      .then(() => console.log('Service worker registrado'))
      .catch(err => console.warn('Service worker failed', err));
  }
}
